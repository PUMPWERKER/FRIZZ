# The filter (#304) on the 5th white key: latched on a 110Hz sine against a MIDI clock; the
# resonance up, the LFO at full depth and 1/16 sweeping it across the centre, lowpass to
# highpass and back; the cutoff to the lowpass side, then the highpass side; page 2's stereo
# up (the right LFO behind); the resonance at the top, the cutoff swept down to the left;
# then off. Nothing is saved. (turn: 4 is knob 1, 1-3 knobs 2-4)
input sine 110 0.5
# what the card keeps, as a fresh card has it: the device's card holds the player's own (#214)
set settings aux_dry 0
set settings aux_live 1
set settings mic_dry 0
set settings mic_live 0
clock 120
at 6000
# latched: its key lit, pink
down KEY_5
wait 80
tap KEY_26
up KEY_5
wait 300
# the resonance up, the LFO in at full depth, 1/16
turn 1 20
wait 300
turn 2 100
wait 300
turn 3 -100
wait 2000
# the cutoff at the centre, white (flat)
expect led pth 1 fdfdfd
# the cutoff to the lowpass side, a little; then right, past the centre
turn 4 -15
wait 2000
turn 4 30
wait 2000
# right of the centre: the cutoff towards orange (highpass)
expect led pth 1 fddcbb
# page 2: stereo up, the right LFO behind, then back
tap ENC_4_SW
wait 300
turn 1 100
wait 1500
turn 1 -100
wait 1000
tap ENC_4_SW
wait 300
# the LFO out, the resonance at the top, the cutoff swept from the right down to the left
turn 2 -100
wait 300
turn 1 100
wait 300
turn 4 -20
wait 500
turn 4 -20
wait 500
turn 4 -20
wait 1000
# far left: blue (lowpass), the resonance at the top
expect led pth 1 1616fd
expect led pth 2 00d1fd
# off again
down KEY_5
wait 80
up KEY_5
wait 1000
# off: its key dim
expect led smt 20 240c14
