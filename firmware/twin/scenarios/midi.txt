# MIDI control (MidiControl.h, #97): notes as keys and the global CCs on channel 16, an
# effect's CCs on its own channel, a program change, a morph
input sine 220 0.3
at 6000
# the global channel 16, whatever the device's card says (in RAM while remote.py plays)
usb F0 7D 43 48 13 00 10 F7
# a note holds the filter, CC 14 on its channel, 5, closes it, the note lets go
midi 9F 37 64
midi B4 0E 10
wait 1500
# a note holds the filter: its knobs on the panel, CC 14 closing knob 1
expect led pth 1 3737fd
expect led pth 2 798fc6
midi 8F 37 00
wait 500
# CC 106 latches the filter, CC 14 on its channel sweeps its cutoff, CC 106 unlatches it
midi BF 6A 7F
midi B4 0E 20
wait 1000
midi B4 0E 60
wait 1000
midi BF 6A 00
wait 500
# the delay latched (CC 113), its knobs 4 and 1 set on its channel, 12; CC 23 puts the
# freezer in focus, and the global CC 14 sets its knob 1
midi BF 71 7F BB 11 60 BB 0E 40
wait 1500
midi BF 17 01 BF 0E 03
wait 500
# the blank scene by program change: every effect off
midi CF 00
wait 1500
# program change 0: the blank scene, its key bright
expect led smt 0 fcfcfc
# notes 45-47: LOOP records, LOOP again plays
midi 9F 2F 64
wait 60
# note 47: LOOP records
expect led pth 8 fd0000
midi 8F 2F 00
wait 2000
midi 9F 2F 64
wait 60
# and plays
expect led pth 6 79f200
midi 8F 2F 00
wait 1500
# the wow & flutter latched (CC 111), then a morph to the blank scene over 2 bars
midi BF 6F 7F BF 09 02 BF 03 00
wait 5000
expect on
